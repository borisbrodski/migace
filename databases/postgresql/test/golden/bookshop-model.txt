view v_order_totals
  definition:
    | SELECT customer_id,
    |    count(*) AS orders,
    |    sum(total_cents) AS total_cents
    |   FROM orders o
    |  GROUP BY customer_id;

function refresh_order_total()
  body:
    |
    |begin
    |  update orders o
    |  set total_cents = (select coalesce(sum(i.quantity * i.price_cents), 0)
    |                     from order_item i
    |                     where i.order_id = o.id)
    |  where o.id = coalesce(new.order_id, old.order_id);
    |  return null;
    |end 
  language: plpgsql
  returns: trigger
