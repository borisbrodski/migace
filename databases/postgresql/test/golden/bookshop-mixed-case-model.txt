trigger OrderItem.TRG_OrderItem_Total
  events: insert, delete, update
  function: RefreshOrderTotal()
  level: row
  timing: after

view V_OrderTotals
  definition:
    | SELECT "CustomerId",
    |    count(*) AS "Orders",
    |    sum("TotalCents") AS "TotalCents"
    |   FROM "Order" o
    |  GROUP BY "CustomerId";

function RefreshOrderTotal()
  body:
    |
    |begin
    |  update "Order" o
    |  set "TotalCents" = (select coalesce(sum(i."Quantity" * i."PriceCents"), 0)
    |                      from "OrderItem" i
    |                      where i."OrderId" = o."Id")
    |  where o."Id" = coalesce(new."OrderId", old."OrderId");
    |  return null;
    |end 
  language: plpgsql
  returns: trigger
